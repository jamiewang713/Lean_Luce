def Luce.rankOf : {n : ℕ} → (Fin n → ℝ) → Fin n → ℕ :=
fun {n} e k => 1 + {j ∈ Finset.univ.erase k | e j < e k}.card
def Luce.otherSurvivors : {n : ℕ} → (Fin n → ℝ) → Fin n → ℝ → ℕ :=
fun {n} e k t => {j ∈ Finset.univ.erase k | t < e j}.card
