Luce.raceRank_eq_iff_survivors {n : ℕ} (times : Fin n → ℝ) (hinj : Function.Injective times) (i : Fin n) (k : ℕ)
  (hk : 1 ≤ k) (hkn : k ≤ n) : Luce.raceRank times i = k ↔ (Luce.survivorSet times (times i)).card = n - k
Luce.exponentialRace_disintegrate {n : ℕ} (w : Luce.Weights (n + 1)) (i : Fin (n + 1)) (g : ℝ × (Fin n → ℝ) → ENNReal)
  (hg : Measurable g) :
  ∫⁻ (clocks : Fin (n + 1) → ℝ), g (clocks i, fun j => clocks (i.succAbove j)) ∂Luce.exponentialRace w =
    ∫⁻ (t : ℝ),
      ProbabilityTheory.exponentialPDF (w.rate i) t *
        ∫⁻ (background : Fin n → ℝ), g (t, background) ∂Luce.backgroundRace w i
Luce.rank_integral_survivors {n : ℕ} (w : Luce.Weights (n + 1)) (i : Fin (n + 1)) (m : ℕ) :
  (Luce.exponentialRace w) {clocks | (Luce.backgroundSurvivors (clocks i) fun j => clocks (i.succAbove j)) = m} =
    ∫⁻ (t : ℝ),
      ProbabilityTheory.exponentialPDF (w.rate i) t *
        (Luce.backgroundRace w i) {background | Luce.backgroundSurvivors t background = m}
Luce.rank_integral_real {n : ℕ} (w : Luce.Weights (n + 1)) (i : Fin (n + 1)) (m : ℕ) :
  (Luce.exponentialRace w).real {clocks | (Luce.backgroundSurvivors (clocks i) fun j => clocks (i.succAbove j)) = m} =
    ∫ (t : ℝ),
      ProbabilityTheory.exponentialPDFReal (w.rate i) t *
        (Luce.backgroundRace w i).real {background | Luce.backgroundSurvivors t background = m}
Luce.raceRank_probability_integral {n : ℕ} (w : Luce.Weights (n + 1)) (i : Fin (n + 1)) (r : ℕ) (hr : 1 ≤ r)
  (hrn : r ≤ n + 1) :
  (Luce.exponentialRace w).real {clocks | Luce.raceRank clocks i = r} =
    ∫ (t : ℝ) in Set.Ioi 0,
      w.rate i * Real.exp (-w.rate i * t) *
        (Luce.exponentialRace w).real {clocks | ((Luce.survivorSet clocks t).erase i).card = n + 1 - r}
Luce.fixed_point_probability_integral {n : ℕ} (w : Luce.Weights (n + 1)) (i : Fin (n + 1)) :
  (Luce.exponentialRace w).real {clocks | Luce.raceRank clocks i = ↑i + 1} =
    ∫ (t : ℝ) in Set.Ioi 0,
      w.rate i * Real.exp (-w.rate i * t) *
        (Luce.exponentialRace w).real {clocks | ((Luce.survivorSet clocks t).erase i).card = n - ↑i}
