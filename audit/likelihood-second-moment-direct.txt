theorem Luce.BernoulliProcess.integral_likelihood_sq_le.{u_1} : ∀ {Ω : Type u_1} {mΩ : MeasurableSpace Ω}
  {μ : Measure Ω} [IsProbabilityMeasure μ] (X : Luce.BernoulliProcess μ) {δ K : ℝ},
  δ < 1 →
    ∀ (g : ℕ → ℝ),
      (∀ (k : ℕ), 0 ≤ g k) →
        (∀ (k : ℕ) (ω : Ω), X.probability k ω ≤ δ) →
          ∀ (N : ℕ),
            (∀ (ω : Ω), ∑ k ∈ Finset.range N, X.probability k ω ≤ K) →
              ∀ {m : ℕ}, m ≤ N → ∫ (ω : Ω), X.likelihood g m ω ^ 2 ∂μ ≤ Real.exp (K / (1 - δ)) :=
fun {Ω} {mΩ} {μ} [IsProbabilityMeasure μ] X {δ K} hδ g hg hpδ N hK {m} hm =>
  have hL := Luce.BernoulliProcess.likelihood_integrable X hδ g hg hpδ m;
  Trans.trans
    (integral_mono (Luce.BernoulliProcess.integrable_likelihood_sq X hδ g hg hpδ N hK hm)
      (Integrable.const_mul hL (Real.exp (K / (1 - δ)))) fun ω =>
      have hb := Luce.BernoulliProcess.likelihood_bounds_of_terminal_cap X hδ g hg hpδ N hK hm ω;
      Eq.mpr
        (id (congrFun' (congrArg LE.le (pow_two (X.likelihood g m ω))) (Real.exp (K / (1 - δ)) * X.likelihood g m ω)))
        (mul_le_mul_of_nonneg_right hb.right hb.left))
    (Eq.mpr
      (id
        (congrArg (fun _a => _a = Real.exp (K / (1 - δ)))
          (integral_const_mul (Real.exp (K / (1 - δ))) (X.likelihood g m))))
      (Eq.mpr
        (id
          (congrArg (fun _a => Real.exp (K / (1 - δ)) * _a = Real.exp (K / (1 - δ)))
            (Luce.BernoulliProcess.integral_likelihood X hδ g hg hpδ m)))
        (Eq.mpr (id (congrArg (fun _a => _a = Real.exp (K / (1 - δ))) (mul_one (Real.exp (K / (1 - δ))))))
          (Eq.refl (Real.exp (K / (1 - δ)))))))
