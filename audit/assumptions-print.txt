@[reducible] def Luce.WeightArray : Type :=
(n : ℕ) → Luce.Weights n
def Luce.NormalizedWeights : Luce.WeightArray → Prop :=
fun w => ∀ (n : ℕ), 0 < n → 1 / ↑n * ∑ i, (w n).rate i = 1
def Luce.stepProfile : Luce.WeightArray → ℕ → ℝ → ℝ :=
fun w n x => ∑ i, if ↑↑i / ↑n < x ∧ x ≤ (↑↑i + 1) / ↑n then (w n).rate i else 0
def Luce.profileMeasure : MeasureTheory.Measure ℝ :=
MeasureTheory.volume.restrict (Set.Ioo 0 1)
def Luce.ProfileL1Convergence : Luce.WeightArray → (ℝ → ℝ) → Prop :=
fun w f =>
  Filter.Tendsto (fun n => MeasureTheory.eLpNorm (fun x => Luce.stepProfile w n x - f x) 1 Luce.profileMeasure)
    Filter.atTop (nhds 0)
def Luce.ProfileLimit : Luce.WeightArray → (ℝ → ℝ) → Prop :=
fun w f =>
  MeasureTheory.NullMeasurable f Luce.profileMeasure ∧ (∀ x ∈ Set.Ioo 0 1, 0 < f x) ∧ Luce.ProfileL1Convergence w f
def Luce.ProfileAssumption : Luce.WeightArray → Prop :=
fun w => ∃ f, Luce.ProfileLimit w f
def Luce.EndpointAssumption : Luce.WeightArray → Prop :=
fun w => ∃ γ ε₀ n₀, 0 < γ ∧ 0 < ε₀ ∧ ∀ (n : ℕ), n₀ ≤ n → ∀ (k : Fin n), (1 - ε₀) * ↑n ≤ ↑↑k + 1 → γ ≤ (w n).rate k
